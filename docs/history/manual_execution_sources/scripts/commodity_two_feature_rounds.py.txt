def validate_inputs(values, names, pairs, round_number):
    import numpy as np
    import pandas as pd
    a = np.asarray(values)
    required = SIGNALS if round_number == 14 else PRIOR_SOURCES
    if (a.ndim != 3 or not 1 <= len(a) <= STOP or a.dtype.kind != 'f'
        or a.shape[2] != len(names) or len(set(names)) != len(names)
        or not set(required).issubset(names) or np.isinf(a).any()):
        raise Stop('Bounded, named, finite-or-missing parent panel required.')
    if (not isinstance(pairs, pd.DataFrame) or pairs.empty or not pairs.columns.is_unique
        or not {'target', 'lag'}.issubset(pairs.columns)
        or pairs[['target','lag']].isna().any().any() or pairs.target.duplicated().any()
        or not pairs.target.map(lambda v:isinstance(v,str) and bool(v)).all()
        or pd.api.types.is_bool_dtype(pairs.lag) or not pairs.lag.isin([1,2,3,4]).all()
        or a.shape[1] != len(pairs)):
        raise Stop('Unique ordered target/horizon metadata required.')
    return a

def ordinal(values):
    import numpy as np
    import pandas as pd
    a = np.array(values, dtype=float, copy=True)
    if a.ndim != 2: raise Stop('Rank input requires two axes.')
    a[~np.isfinite(a)] = np.nan
    n = np.isfinite(a).sum(axis=1)
    q = pd.DataFrame(a).rank(axis=1, method='average').to_numpy(dtype=float, copy=True)
    result = 2*(q-1)/np.maximum(n[:,None]-1,1)-1
    result[n<3] = np.nan
    return result

def by_horizon(values, pairs):
    import numpy as np
    h = pairs.lag.to_numpy(dtype=int, copy=True)
    result = np.full_like(values, np.nan, dtype=float)
    for k in (1,2,3,4):
        idx = np.flatnonzero(h==k)
        if len(idx): result[:,idx] = ordinal(values[:,idx])
    return result

def historical_surprise(values, robust=False):
    import numpy as np
    import pandas as pd
    a = np.array(values, dtype=float, copy=True)
    h = pd.DataFrame(a).shift(1).rolling(63, min_periods=42)
    if robust:
        loc = h.median().to_numpy(dtype=float,copy=True)
        scale = (h.quantile(.75)-h.quantile(.25)).to_numpy(dtype=float,copy=True)
    else:
        loc = h.mean().to_numpy(dtype=float,copy=True)
        scale = h.std(ddof=0).to_numpy(dtype=float,copy=True)
    good = np.isfinite(a)&np.isfinite(loc)&np.isfinite(scale)&(scale>1e-12)
    out = np.full_like(a,np.nan)
    with np.errstate(over='ignore',invalid='ignore',divide='ignore'):
        np.divide(a-loc,scale,out=out,where=good)
    out[~np.isfinite(out)] = np.nan
    return out

def safe_ratio(num, den):
    import numpy as np
    a,b = np.asarray(num,float),np.asarray(den,float)
    if a.shape!=b.shape: raise Stop('Ratio axes differ.')
    out = np.full_like(a,np.nan)
    with np.errstate(over='ignore',invalid='ignore',divide='ignore'):
        np.divide(a,b,out=out,where=np.isfinite(a)&np.isfinite(b)&(b>1e-12))
    out[~np.isfinite(out)] = np.nan
    return out

def feature_block(values, names, pairs, round_number):
    import numpy as np
    import pandas as pd
    info(round_number)
    a = validate_inputs(values,names,pairs,round_number)
    channels, meta = [], []
    def add(v, name, group, source, form, inherited=True, reused=False):
        b = np.array(v,dtype=float,copy=True)
        b[~np.isfinite(b)] = np.nan
        channels.append(b)
        meta.append(dict(name=name,group=group,source=source,form=form,
            uses_released_label_inputs=inherited,reused_notebook13_representation=reused))
    if round_number==14:
        for source in SIGNALS:
            raw=a[:,:,names.index(source)]
            for robust,group in [(False,'ordinary'),(True,'robust')]:
                z=historical_surprise(raw,robust)
                prefix='innovation14__'+source.replace('__','_')+'__'+group
                add(np.clip(z,-12,12),prefix+'_z',group,source,'magnitude',source.startswith('released_priors__'))
                # Rank BEFORE clipping so previous notebook-13 ranks remain exactly reproducible.
                add(by_horizon(z,pairs),prefix+'_rank',group,source,'rank',source.startswith('released_priors__'),not robust)
    else:
        q={s.removeprefix('released_priors__'):np.array(a[:,:,names.index(s)],float,copy=True) for s in PRIOR_SOURCES}
        drivers=[safe_ratio(q['location_63']-q['location_126'],q['risk_126']),
                 safe_ratio(q['location_126']-q['location_252'],q['risk_126']),
                 safe_ratio(q['median_63']-q['location_63'],q['risk_63']),
                 safe_ratio(q['location_126']-q['market_pair_pool'],q['risk_126']),
                 safe_ratio(q['location_126']-q['canonical_cross_horizon_pool'],q['risk_126'])]
        ratio=safe_ratio(q['risk_63'],q['risk_252'])
        risk=np.full_like(ratio,np.nan)
        np.log(ratio,out=risk,where=np.isfinite(ratio)&(ratio>0))
        drivers.append(risk)
        for j,(mechanism,raw) in enumerate(zip(MECHANISMS,drivers,strict=True)):
            group='own_history' if j<3 else 'pool_and_risk'
            forms=[np.clip(raw,-12,12),np.clip(pd.DataFrame(raw).diff(5).to_numpy(dtype=float,copy=True),-12,12),
                   np.clip(historical_surprise(raw),-12,12),by_horizon(raw,pairs)]
            for form,v in zip(['level','change_5','surprise_63','horizon_rank'],forms,strict=True):
                add(v,'prior_dynamics15__'+mechanism+'__'+form,group,mechanism,form)
    output=np.stack(channels,axis=2).astype(np.float32)
    output_names=[r['name'] for r in meta]
    if output.shape!=(len(a),len(pairs),24) or np.isinf(output).any() or len(set(output_names))!=24:
        raise Stop('Invalid 24-template feature inventory.')
    return output,output_names,meta

def audit_candidates(base, cube, meta, pairs, truth, audit, old_cube=None, old_names=None):
    """Training-only diagnostics, NOT outcome-driven panel re-selection."""
    import numpy as np
    if len(base.values)<TRAIN_STOP or len(cube)<TRAIN_STOP or np.asarray(truth).shape!=(TRAIN_STOP,len(pairs)):
        raise Stop('Expected only the middle-fold fitting prefix for diagnostics.')
    lo,mid,hi=252,798,TRAIN_STOP
    baseline={audit.signature(base.values[lo:hi,:,j]):n for j,n in enumerate(base.names)}
    seen=dict(baseline)
    old={} if old_cube is None else {audit.signature(old_cube[252:1164,:,j]):n for j,n in enumerate(old_names)}
    rows=[]
    for j,m in enumerate(meta):
        data=cube[lo:hi,:,j]
        sig=audit.signature(data);duplicate=seen.get(sig)
        if duplicate is None:seen[sig]=m['name']
        daily=audit.daily_ic(cube[:hi,:,j],truth)
        means=[];counts=[]
        for p in (daily[lo:mid],daily[mid:hi]):
            z=p[np.isfinite(p)];counts.append(len(z));means.append(float(z.mean()) if len(z) else None)
        finite=data[np.isfinite(data)]
        sample=finite[::max(1,len(finite)//10000)]
        hist,edges=np.histogram(sample,bins=24) if len(sample) else (np.zeros(24,int),np.linspace(-1,1,25))
        rows.append(dict(**m,training_coverage=float(np.isfinite(data).mean()),
            half_1_mean_ic=means[0],half_2_mean_ic=means[1],half_1_dates=counts[0],half_2_dates=counts[1],
            same_sign=bool(means[0] is not None and means[1] is not None and means[0]*means[1]>0),
            duplicate_of=duplicate,notebook13_duplicate=old.get(audit.signature(cube[252:1164,:,j])),
            training_sha256=sig,median=float(np.median(finite)) if len(finite) else None,
            q05=float(np.quantile(finite,.05)) if len(finite) else None,
            q95=float(np.quantile(finite,.95)) if len(finite) else None,
            histogram=dict(counts=hist.tolist(),edges=edges.tolist(),sampled_entries=len(sample))))
    return rows

def smoke(round_number):
    import numpy as np
    import pandas as pd
    rng=np.random.default_rng(20260912)
    names=list(dict.fromkeys(SIGNALS+PRIOR_SOURCES))
    a=rng.normal(0,.1,(150,16,len(names))).astype(np.float32)
    for j,n in enumerate(names):
        if 'risk_' in n or n.endswith('risk_63_difference'):a[:,:,j]=np.abs(a[:,:,j])+.01
    pairs=pd.DataFrame(dict(target=[f'target_{i}' for i in range(16)],lag=np.tile([1,2,3,4],4)))
    original=a.copy();full,n,meta=feature_block(a,names,pairs,round_number)
    changed=a.copy();changed[100:]*=3
    other,_,_=feature_block(changed,names,pairs,round_number)
    np.testing.assert_array_equal(full[:100],other[:100])
    for cut in (75,121):
        b,nn,_=feature_block(a[:cut],names,pairs,round_number)
        np.testing.assert_array_equal(full[:cut],b)
        if n!=nn:raise Stop('Prefix changed feature names.')
    np.testing.assert_array_equal(a,original);variants(round_number,meta)
    if not np.isfinite(full[100:]).all():raise Stop('Synthetic complete history unexpectedly unavailable.')
    return dict(status='TWO_ROUND_CONTRACT_PASSED',round=round_number,input_unchanged=True,
        future_input_perturbation_exact=True,prefix_exact=True,templates=24,
        pandas_version=pd.__version__,numpy_version=np.__version__)

def fit_one(root,u,round_number,name,chosen,stage,lineage,base,cube,names,y,pairs,fold,config,r):
    import numpy as np
    import pandas as pd
    import joblib
    from commodity_prediction.domain.attribution.run import fit_stage
    from commodity_prediction.domain.experiment import Experiment
    from commodity_prediction.domain.model import prepare,select
    if stage.exists():raise Stop('Unexpected model stage preserved; no automatic refit.')
    panel=append_panel(base,cube,names,chosen)
    settings={**config,'max_features':len(panel.names),'max_abs_correlation':1.01}
    stats=prepare(panel,y,fold.train_stop,settings)
    exp=Experiment(name,algorithm='histogram')
    _,screen=select(stats,exp.candidates(panel.names),settings,False)
    admitted=[n for n in chosen if n in screen['selected_names']]
    if set(screen['rejection_reasons'])&{'feature_budget','correlated','unstable_sign'}:
        raise Stop('Unexpected fit-time screening rule.')
    if not admitted:
        # A valid negative screen does not waste a model fit or abort the other panels.
        r['skipped_panels'].append(dict(variant=name,reason='No new nonconstant, nonduplicate, sufficiently observed input',selection=screen))
        u.atomic_json(report_path(root,round_number),r)
        return
    if r['fit_attempts_this_call']>=4:raise Stop('Four-fit limit reached.')
    r['fit_attempts_this_call']+=1;u.atomic_json(report_path(root,round_number),r)
    result=fit_stage(panel,y,pairs,fold,exp,settings,stats,stage,lineage)
    saved=pd.read_parquet(stage/'predictions.parquet')
    replay=joblib.load(stage/'model.joblib').predict(panel,START,STOP)
    error=u.exact_predictions(saved,replay)
    r['results'].append({**result,'new_inputs_admitted':len(admitted),'intended_inputs':chosen})
    r['new_training_fits']+=1
    r['maximum_prediction_replay_error']=max(error,r['maximum_prediction_replay_error'])
    r['checkpoint_hashes'][name]=u.verify_stage(stage,lineage)
    u.atomic_json(report_path(root,round_number),r)
    emit('FEATURE_MODEL_CHECKPOINTED',round=round_number,variant=name,completed=r['new_training_fits'],limit=4,
        official_metric=result['metrics']['official_metric'],new_inputs=len(admitted))
    del panel,stats,replay;gc.collect()

def comparison_rows(round_number,results,control,groups,metric):
    import numpy as np
    ref=metric(control['daily_rank_correlations'],180)
    if not math.isclose(ref,control['official_metric'],rel_tol=0,abs_tol=1e-12):raise Stop('Control metric differs from daily series.')
    if len({r['variant'] for r in results})!=len(results):raise Stop('Duplicate model results.')
    scores={'current_market':ref}
    for r in results:
        v=r['variant']

        scores[v]=score
    if round_number==14:
        references={'audit_shortlist':[],'ordinary_innovations':[],
            'robust_innovations':['ordinary_innovations'],'innovation_joint':['ordinary_innovations','robust_innovations']}
    else:

            'pool_and_risk_dynamics':['risk_dynamics_only'],'prior_dynamics_joint':['own_history_dynamics','pool_and_risk_dynamics']}
    rows=[];qualified=[]
    for r in results:
        v=r['variant'];chosen=r['selection']['selected_names']
        admitted=[n for n in groups[v] if n in chosen]
        if r.get('intended_inputs')!=groups[v] or r['new_inputs_admitted']!=len(admitted):raise Stop('Feature admission counts disagree.')
        all_admitted=set(groups[v]).issubset(chosen)
        refs=['current_market']+references[v]
        complete=all(k in scores for k in refs)
        deltas={k:scores[v]-scores[k] for k in refs if k in scores}
        passed=complete and all_admitted and all(d>=GAIN for d in deltas.values())
        rows.append(dict(variant=v,official_metric=scores[v],delta_vs_current_market=scores[v]-ref,
            declared_comparator_deltas=deltas,all_comparators_available=complete,
            intended_inputs=len(groups[v]),new_inputs_admitted=len(admitted),
            all_intended_inputs_admitted=all_admitted,passes_review_gate=bool(passed)))
        if passed:qualified.append(v)
    return rows,qualified

    # Shared lock forbids parallel rounds competing for the 4-core CPU/memory.
    with u.safe_path(root,'logs/manual_two_feature_rounds.lock').open('a') as lock:
        fcntl.flock(lock.fileno(),fcntl.LOCK_EX|fcntl.LOCK_NB)
        path=report_path(root,round_number)
        if path.exists():
            r=u.read_json(path)
            if r.get('status') not in TERMINAL:raise Stop('Interrupted round exists; no automatic retry or time reset. Preserve its report.')
            verify_complete(root,r,audit,session,u);emit('COMPLETED_ROUND_REUSED',round=round_number,new_fits=0)
            return {**r,'new_fits_this_notebook_call':0}
        lineage,_=identity(root,round_number,u)
        if (root/'artifacts/two_feature_rounds'/info(round_number)['slug']/lineage).exists():raise Stop('Orphan artifacts preserved; no refit.')
        if shutil.disk_usage(root).free<1024**3:raise Stop('At least 1 GiB free required. Do not delete old projects blindly.')

        cmd=[ready['runtime']['python'],'-u',str(Path(__file__).resolve()),'--worker','--round',str(round_number),'--root',str(root)]
        began=time.monotonic()
        try:session._supervise_command(cmd,root,LIMIT,out/'worker.log',u.environment(root))
        except BaseException as exc:
            r=u.read_json(path)
            if r.get('status')=='STARTING_WORKER':r['new_training_fits']=None
            r.update(status='STOPPED',supervisor_error=type(exc).__name__+': '+str(exc),supervised_seconds=round(time.monotonic()-began,3))
            u.atomic_json(path,r);raise
        r=u.read_json(path);verify_complete(root,r,audit,session,u)
        r.update(supervised_seconds=round(time.monotonic()-began,3),new_fits_this_notebook_call=r['new_training_fits'])

def backup(root,r,u):
    rnd=r['round'];d=root/'artifacts/two_feature_rounds'/info(rnd)['slug']/r['lineage']

    pins={str(p.relative_to(root)):u.digest(u.safe_path(root,str(p.relative_to(root)))) for p in files}
    if sum(p.stat().st_size for p in files)>256*1024**2:raise Stop('Backup larger than 256 MiB; results preserved.')
    dest=u.safe_path(outdir(root,rnd),info(rnd)['slug']+'_checkpoints.zip');tmp=u.safe_path(dest.parent,dest.name+'.tmp')
    with zipfile.ZipFile(tmp,'w',zipfile.ZIP_DEFLATED,compresslevel=1) as z:
        for p in files:z.write(p,str(p.relative_to(root)))
        z.writestr('SHA256SUMS.json',json.dumps(pins,sort_keys=True,indent=2))
    with zipfile.ZipFile(tmp) as z:
        if z.testzip():raise Stop('Backup CRC failed.')
        for n,h in pins.items():
            if hashlib.sha256(z.read(n)).hexdigest()!=h:raise Stop('Backup read-back checksum differs.')
    os.replace(tmp,dest)
    return dict(path=str(dest),sha256=u.digest(dest),bytes=dest.stat().st_size,files=len(files),private=True,
        label_derived_features_models_predictions=True,raw_csvs_included=False,off_disk_backup_confirmed=False)

def same_notebook_sources(a,b):
    def source(c):return ''.join(c['source']) if isinstance(c['source'],list) else c['source']
    return [(c['cell_type'],source(c)) for c in a['cells']]==[(c['cell_type'],source(c)) for c in b['cells']]

def install(root=ROOT):
    root=Path(root);audit,session,older,u=previous(root);u.validate_source(root);read_audit(root,u)
    emit('INSTALL_TWO_INDEPENDENT_ROUNDS',new_training_fits=0)
    payloads={u.safe_path(root,'scripts/commodity_two_feature_rounds.py'):Path(__file__).read_bytes()}
    notebooks={}
    for rnd in (14,15):
        i=info(rnd);nb=notebook_document(rnd);p=u.safe_path(root,'notebooks/'+i['notebook']);notebooks[p]=nb
        payloads[p]=(json.dumps(nb,indent=1)+'\n').encode()
        payloads[u.safe_path(root,'configs/manual_'+i['slug']+'.json')]=(json.dumps(plan(rnd),sort_keys=True,indent=2)+'\n').encode()
        payloads[u.safe_path(root,'docs/manual_'+i['slug']+'.md')]=protocol(rnd).encode()
    for p,body in payloads.items():
        if p.exists():
            if p in notebooks:
                if not same_notebook_sources(u.read_json(p),notebooks[p]):raise Stop('Notebook edits preserved: '+str(p))
            elif p.read_bytes()!=body:raise Stop('Existing differing file preserved: '+str(p))
    for p,body in payloads.items():
        if not p.exists():u.write_new(p,body)
    print('RESULT: TWO_FEATURE_ROUNDS_READY',flush=True)
    for rnd in (14,15):print(f'ROUND_{rnd}: '+str(root/'notebooks'/info(rnd)['notebook']),flush=True)
    print('KERNEL: Commodity - manual (verified)\nRun sequentially, never simultaneously. Setup performed zero fits.',flush=True)
    return dict(new_training_fits=0,files=[str(p) for p in payloads])

def main():
    parser=argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--root',type=Path,default=ROOT)
    parser.add_argument('--worker',action='store_true',help=argparse.SUPPRESS)
    parser.add_argument('--round',type=int,choices=[14,15])
    parser.add_argument('--run-round',type=int,choices=[14,15],help='Terminal alternative in the verified Python environment; runs and exports ONE round.')
    args=parser.parse_args();old=None
    try:
        if args.worker:
            if args.round is None:raise Stop('Worker requires explicit round.')
            worker(args.root,args.round)
        elif args.run_round:
            if args.round is not None:raise Stop('Do not combine --round and --run-round.')
            r=run(args.root,args.run_round);r=finish(args.root,r,charts(r))
            print('RESULT:',r['status']);print('REPORT:',report_path(args.root,args.run_round))
        else:
            if args.round is not None:raise Stop('Use a notebook or --run-round; no implicit fit.')
            def expire(*_):raise Stop('90-second setup limit reached.')
            old=signal.signal(signal.SIGALRM,expire);signal.alarm(90);install(args.root)
    except BaseException as exc:
        print('RESULT: STOPPED\nERROR: '+type(exc).__name__+': '+str(exc),flush=True)
        print('Preserve report/log. Do not repeat unchanged. Stop space manually; never delete it.',flush=True)
        return 2
    finally:
        if old is not None:signal.alarm(0);signal.signal(signal.SIGALRM,old)
    return 0
