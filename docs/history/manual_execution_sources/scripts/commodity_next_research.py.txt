def clean_labels(labels,pairs,*other):
    import numpy as np
    y=np.array(labels,dtype=float,copy=True)
    if y.ndim!=2 or not 1<=len(y)<=1704:raise p.Stop('Expected bounded development outcome matrix.')
    p.bounded_inputs(np.zeros((*y.shape,1),np.float32),['axis'],pairs,['axis'])
    for value in other:
        if np.asarray(value).shape!=y.shape:raise p.Stop('Outcome and reference axes differ.')
    y[(~np.isfinite(y))|(y==-999999)]=np.nan
    return y


def release_stream(values,pairs):
    import numpy as np
    a=np.array(values,float,copy=True); out=np.full_like(a,np.nan)
    for j,h in enumerate(pairs.lag):
        delay=int(h)+1
        if len(a)>delay:out[delay:,j]=a[:-delay,j]
    return out


def clocks(valid,window,minimum):
    """Coverage plus mean age of actually observed release rows, never magnitudes."""
    import numpy as np
    import pandas as pd
    good=np.asarray(valid,bool); dates=np.arange(len(good),dtype=float)[:,None]
    cnt=pd.DataFrame(good.astype(float)).rolling(window,min_periods=1).sum().to_numpy(copy=True)
    stamped=np.where(good,np.broadcast_to(dates,good.shape),np.nan)
    mean_date=pd.DataFrame(stamped).rolling(window,min_periods=minimum).mean().to_numpy(copy=True)
    cover=cnt/window; cover[np.arange(len(good))<minimum-1]=np.nan
    age=np.log1p(np.maximum(0,dates-mean_date)/window)
    return cover,age


def ratio(a,b):
    return p.ratio(a,b)


def sequence_block(labels,risk,pairs):
    """24 descriptors of legally released, h-spaced pairs. No outcome imputation."""
    import numpy as np
    import pandas as pd
    y=clean_labels(labels,pairs,risk);known=release_stream(y,pairs);lag=np.full_like(known,np.nan)
    for j,h in enumerate(pairs.lag):
        h=int(h)
        if len(y)>h:lag[h:,j]=known[:-h,j]
    valid=np.isfinite(known)&np.isfinite(lag); x=np.where(valid,known,np.nan);q=np.where(valid,lag,np.nan)
    arrays=[];meta=[]
    for w in WINDOWS:
        minimum=max(8,w*2//3)
        def mean(z):return pd.DataFrame(z).rolling(w,min_periods=minimum).mean().to_numpy(copy=True)
        def count(mask):return pd.DataFrame(mask.astype(float)).rolling(w,min_periods=1).sum().to_numpy(copy=True)
        mx,mq=mean(x),mean(q)
        vx=np.maximum(mean(x*x)-mx*mx,0);vq=np.maximum(mean(q*q)-mq*mq,0);cov=mean(x*q)-mx*mq
        corr=ratio(cov,np.sqrt(vx*vq));beta=ratio(cov,vq)
        corr=np.clip(corr,-1,1)
        nz=valid&(known!=0)&(lag!=0)
        agree_count=count(nz&(np.sign(known)==np.sign(lag)));nz_count=count(nz)
        agreement=ratio(agree_count,nz_count);agreement[nz_count<minimum]=np.nan
        signed=np.sign(known)*(2*agreement-1)
        innovation=ratio(known-(mx+beta*(lag-mq)),np.asarray(risk,float))
        pos=valid&(lag>0);neg=valid&(lag<0)
        pc,nc=count(pos),count(neg);allc=count(valid)
        up=ratio(count(valid&(known>0)),allc);down=ratio(count(valid&(known<0)),allc)
        up_p=ratio(count(pos&(known>0)),pc);up_n=ratio(count(neg&(known>0)),nc)
        down_n=ratio(count(neg&(known<0)),nc)
        adequate=(allc>=minimum)&(pc>=3)&(nc>=3)
        positive=up_p-up;negative=down_n-down
        response=np.where(known>0,up_p-up,np.where(known<0,up_n-up,0.0))
        for z in (positive,negative,response):z[~adequate|~np.isfinite(known)]=np.nan
        cover,age=clocks(valid,w,minimum)
        fields=[cover,age,corr,signed,innovation,response,positive,negative]
        for idx,(form,a) in enumerate(zip(FORMS18,fields,strict=True)):
            out=np.array(a,float,copy=True);out[~np.isfinite(out)]=np.nan
            arrays.append(np.clip(out,-12,12))
            meta.append(dict(name=f'released_sequence18__w{w}__{form}',window=w,form=form,
                group='clock' if idx<2 else ('magnitude' if idx<5 else 'transitions'),uses_released_labels=True))
    cube=np.stack(arrays,axis=2).astype(np.float32)
    return cube,[v['name'] for v in meta],meta


def error_block(labels,source_mean,source_risk,pairs):
    """Residuals reference the causal prior at the ORIGINAL forecast origin s."""
    import numpy as np
    import pandas as pd
    y=clean_labels(labels,pairs,source_mean,source_risk)
    # All three terms below refer to origin s. The *whole* error is then delayed.
    reference_error=ratio(y-np.asarray(source_mean,float),np.asarray(source_risk,float))
    error=release_stream(np.clip(reference_error,-12,12),pairs)
    valid=np.isfinite(error);arrays=[];meta=[]
    for w in WINDOWS:
        minimum=max(8,w*2//3);hist=pd.DataFrame(error).rolling(w,min_periods=minimum)
        avg=hist.mean().to_numpy(copy=True);med=hist.median().to_numpy(copy=True)
        sign=pd.DataFrame(np.where(valid,np.sign(error),np.nan)).rolling(w,min_periods=minimum).mean().to_numpy(copy=True)
        absolute=pd.DataFrame(np.abs(error)).rolling(w,min_periods=minimum).mean().to_numpy(copy=True)
        rms=np.sqrt(pd.DataFrame(error*error).rolling(w,min_periods=minimum).mean().to_numpy(copy=True))
        exceed=pd.DataFrame(np.where(valid,(np.abs(error)>2).astype(float),np.nan)).rolling(w,min_periods=minimum).mean().to_numpy(copy=True)
        cover,age=clocks(valid,w,minimum)
        for idx,(form,a) in enumerate(zip(FORMS19,[cover,age,avg,med,sign,absolute,rms,exceed],strict=True)):
            out=np.array(a,float,copy=True);out[~np.isfinite(out)]=np.nan;arrays.append(np.clip(out,-12,12))
            meta.append(dict(name=f'prior_error19__w{w}__{form}',window=w,form=form,
              group='clock' if idx<2 else ('bias' if idx<5 else 'dispersion'),uses_released_labels=True))
    cube=np.stack(arrays,axis=2).astype(np.float32)
    return cube,[v['name'] for v in meta],meta

def identity(root,n):
    if p.read_json(p.safe(root,'configs/manual_'+INFO[n]['slug']+'.json'))!=plan(n):raise p.Stop('Declared experiment changed.')
    b=dict(plan=plan(n),helper_sha256=p.digest(Path(__file__)),shared_helper_sha256=p.digest(Path(p.__file__)),
      tests_sha256=p.digest(tests_path(root)),domain_config_sha256=p.digest(root/'configs/domain_study.json'),
      dependencies=p.PINS)

def test_gate(root):
    p.test_gate(root)
    r=p.read_json(gate_path(root));t=p.read_json(gate_path(root).parent/'tests.json')

       or r.get('shared_helper_sha256')!=p.digest(Path(p.__file__))
       or r.get('tests_sha256')!=p.digest(tests_path(root))
       or r.get('tests_receipt_sha256')!=p.digest(gate_path(root).parent/'tests.json')
       or t.get('status')!='TESTS_PASSED' or any(t.get(k,1) for k in ('failures','errors','skipped'))):
        raise p.Stop('New-round preflight is missing, failed, or stale.')

def preflight(root=ROOT):
    root=Path(root);p.test_gate(root);out=gate_path(root).parent;out.mkdir(parents=True,exist_ok=True)
    if gate_path(root).exists():
        r=test_gate(root);p.emit('NEXT_PREFLIGHT_PASSED',reused=True);return r
    r=dict(status='RUNNING',started_utc=p.utc(),helper_sha256=p.digest(Path(__file__)),
      shared_helper_sha256=p.digest(Path(p.__file__)),tests_sha256=p.digest(tests_path(root)),
      private_model_fits=0,synthetic_forecast_model_fits=0)
    p.atomic_json(gate_path(root),r);began=time.monotonic()
    try:
        with p.time_limit(100,'Next-round preflight deadline.'):
            _,_,_,u=p.previous(root);ready=u.readiness(root);u.validate_runtime(root,ready);u.validate_source(root)
            env=u.environment(root);env['PYTHONPATH']=str(root/'scripts')+os.pathsep+str(root/'src')
            p.supervise([ready['runtime']['python'],'-u',str(tests_path(root)),'--receipt',str(out/'tests.json')],root,80,out/'tests.log',env)
            t=p.read_json(out/'tests.json')
            if t.get('status')!='TESTS_PASSED' or t.get('tests_run',0)<25:raise p.Stop('Prepared tests did not pass.')
            r.update(status='NEXT_PREFLIGHT_PASSED',tests_receipt_sha256=p.digest(out/'tests.json'),
                     elapsed_seconds=round(time.monotonic()-began,3),test_results=t)
            p.atomic_json(gate_path(root),r);p.emit('NEXT_PREFLIGHT_PASSED',tests=t['tests_run']);return r
    except BaseException as exc:
        r.update(status='STOPPED',error=type(exc).__name__+': '+str(exc),traceback=traceback.format_exc())
        p.atomic_json(gate_path(root),r);raise

def aggregate(n,r,config):
    import numpy as np
    from commodity_prediction.studies.evaluation import compare_predictions
    v=INFO[n]['variants'];data={'current_market':r['control']}
    data.update({name:x['metrics'] for name,x in r['results'].items()})
    contrasts=[(name,ref) for name in v if name in data for ref in plan(n)['comparators'][name] if ref in data]
    r['comparisons']=compare_predictions({k:np.asarray(z['daily_rank_correlations']) for k,z in data.items()},[175],contrasts,config) if contrasts else []
    rows=[];clock=v[0]
    for name in v:
        if name not in data:continue
        refs=plan(n)['comparators'][name];deltas={k:data[name]['official_metric']-data[k]['official_metric'] for k in refs if k in data}
        selected=r['results'][name]['selection']['selected_names'];chosen=r['panels'][name]
        numeric=[x for x in chosen if x not in r['panels'][clock]]
        admitted=sum(x in selected for x in chosen);nadmitted=sum(x in selected for x in numeric)
        passing=(len(deltas)==len(refs) and all(x>=.002 for x in deltas.values()) and admitted>0 and (not numeric or nadmitted>0))
        rows.append(dict(variant=name,official_metric=data[name]['official_metric'],delta_vs_current_market=deltas['current_market'],
            comparator_deltas=deltas,intended=len(chosen),admitted=admitted,numerical_admitted=nadmitted,
            exact_dropped_inputs=[x for x in chosen if x not in selected],passes_review_gate=bool(passing)))
    r['comparison_rows']=rows;r['candidates_for_review']=[x['variant'] for x in rows if x['passes_review_gate']]
    r['decision']='REVIEW_ON_OTHER_PERIODS' if r['candidates_for_review'] else 'STOP_TESTED_PANELS'
    r['pooled_score_promoted']=False


def verify_complete(root,n,r):
    lineage,ident=identity(root,n)
    if (r.get('status') not in DONE or r.get('identity')!=ident or r.get('lineage')!=lineage
       or not r.get('parents_unchanged') or not r.get('source_unchanged')
       or r.get('maximum_prediction_replay_error')!=0 or r.get('control_refits')!=0):raise p.Stop('Not a completed compatible round.')
    d=root/'artifacts/manual_next_research'/INFO[n]['slug']/lineage
    p.verify(d/'features',lineage,r['feature_hashes'])
    if set(r['results'])|set(r['skipped'])!=set(INFO[n]['variants']) or set(r['results'])&set(r['skipped']):raise p.Stop('Incomplete task inventory.')
    if r['new_training_fits']!=len(r['results']) or r['fit_attempts_total']!=len(r['results']):raise p.Stop('Fit count differs from sealed inventory.')
    for name,z in r['results'].items():
        p.verify(d/name,lineage,r['stage_hashes'][name])
        if p.read_json(d/name/'result.json')!=z:raise p.Stop('Model report differs.')

    except BaseException as exc:
        r.update(status='STOPPED',error=type(exc).__name__+': '+str(exc),traceback=traceback.format_exc());save();raise


def run(root=ROOT,round_number=18,resume=False,pause_after=None):
    n=round_number;require(n);root=Path(root);test_gate(root)
    if pause_after is not None and (type(pause_after) is not int or pause_after<1):raise p.Stop('Pause count must be positive.')
    if shutil.disk_usage(root).free<2*1024**3:raise p.Stop('At least 2 GiB free disk required; do not delete old work.')
    with p.safe(root,'logs/manual_two_feature_rounds.lock').open('a') as lock:
        try:fcntl.flock(lock.fileno(),fcntl.LOCK_EX|fcntl.LOCK_NB)
        except BlockingIOError:raise p.Stop('Another research worker is active.')
        lineage,ident=identity(root,n);path=report_path(root,n);outdir(root,n).mkdir(parents=True,exist_ok=True)
        if path.exists():
            r=p.read_json(path)
            if r.get('status') in DONE:return verify_complete(root,n,r)
            if not resume or r.get('status')!='PAUSED_AFTER_CHECKPOINT' or r.get('identity')!=ident:raise p.Stop('Only intentional pauses can resume; errors require diagnosis.')
        else:
            if resume:raise p.Stop('No saved pause to resume.')
            if (root/'artifacts/manual_next_research'/INFO[n]['slug']/lineage).exists():raise p.Stop('Orphan artifacts preserved.')
            r=dict(project='commodity-prediction',round=n,status='STARTING_WORKER',source_commit=p.SHA,lineage=lineage,identity=ident,plan=plan(n),
              results={},skipped={},stage_hashes={},completed_tasks=0,new_training_fits=0,fit_attempts_total=0,
              cumulative_supervised_seconds=0.,supervisor_calls=0,control_refits=0,final_test_evaluations=0,
              github_writes=False,aws_api_calls=0,feature_gate='open',promotion_allowed=False,started_utc=p.utc())
        old=float(r['cumulative_supervised_seconds']);remaining=240-old
        if not math.isfinite(remaining) or remaining<10:raise p.Stop('No adequate remaining allowance; budget not reset.')
        r['supervisor_calls']+=1;p.atomic_json(path,r)
        _,_,_,u=p.previous(root);ready=u.readiness(root);env=u.environment(root)
        env['PYTHONPATH']=str(root/'scripts')+os.pathsep+str(root/'src')
        cmd=[ready['runtime']['python'],'-u',str(Path(__file__).resolve()),'--worker','--round',str(n),'--root',str(root),'--remaining',str(remaining)]
        if pause_after is not None:cmd+=['--pause-after',str(pause_after)]
        began=time.monotonic()
        try:
            elapsed=p.supervise(cmd,root,remaining,outdir(root,n)/f'worker_{r["supervisor_calls"]:02d}.log',env,path)
            r=p.read_json(path);r['cumulative_supervised_seconds']=old+elapsed;p.atomic_json(path,r)
            return r if r['status']=='PAUSED_AFTER_CHECKPOINT' else verify_complete(root,n,r)
        except BaseException as exc:
            r=p.read_json(path);r.update(status='STOPPED',supervisor_error=type(exc).__name__+': '+str(exc),
               cumulative_supervised_seconds=old+time.monotonic()-began);p.atomic_json(path,r);raise

def export(root,n,r,figures):
    root=Path(root);verify_complete(root,n,r);out=outdir(root,n)
    with p.time_limit(90,'Presentation deadline; no refitting is needed.'):
        html=out/(INFO[n]['slug']+'_dashboard.html')
        html.write_text('<!doctype html><meta charset="utf-8"><h1>'+INFO[n]['title']+'</h1><p>Development research only.</p>'+''.join(f.to_html(full_html=False,include_plotlyjs=(i==0)) for i,f in enumerate(figures)),encoding='utf-8')
        d=root/'artifacts/manual_next_research'/INFO[n]['slug']/r['lineage'];items={}
        for stage in [d/'features']+[d/x for x in r['stage_hashes']]:
            man=p.read_json(stage/'manifest.json')
            for name in list(man['files'])+['manifest.json']:
                file=p.safe(stage,name);items[str(file.relative_to(d))]=file
        if sum(f.stat().st_size for f in items.values())>250*1024**2:raise p.Stop('Backup exceeds bounded size.')
        dest=out/(INFO[n]['slug']+'_checkpoints.zip');temp=dest.with_suffix('.zip.tmp')
        with zipfile.ZipFile(temp,'w',compression=zipfile.ZIP_DEFLATED,compresslevel=3) as z:
            for rel,file in items.items():z.write(file,rel)
        with zipfile.ZipFile(temp) as z:
            for rel,file in items.items():
                if hashlib.sha256(z.read(rel)).hexdigest()!=p.digest(file):raise p.Stop('ZIP verification failed.')
        os.replace(temp,dest)
        r.update(status='NOTEBOOK_COMPLETE',plotly_figures=len(figures),notebook=str(root/'notebooks'/INFO[n]['notebook']),dashboard=str(html),
          checkpoint_bundle=dict(path=str(dest),sha256=p.digest(dest),private=True,off_disk_backup=False),dashboard_sha256=p.digest(html))
        p.atomic_json(report_path(root,n),r);return r


def main():
    parser=argparse.ArgumentParser();parser.add_argument('--root',type=Path,default=ROOT);parser.add_argument('--round',type=int,choices=[18,19],default=18)
    acts=parser.add_mutually_exclusive_group(required=True);acts.add_argument('--preflight',action='store_true');acts.add_argument('--worker',action='store_true');acts.add_argument('--run',action='store_true');acts.add_argument('--export',action='store_true')
    parser.add_argument('--remaining',type=float);parser.add_argument('--pause-after',type=int);parser.add_argument('--resume',action='store_true');a=parser.parse_args()
    sys.path.insert(0,str(a.root/'src'))
    for key in ('OMP_NUM_THREADS','MKL_NUM_THREADS','OPENBLAS_NUM_THREADS'):os.environ[key]='4'
    try:
        if a.preflight:preflight(a.root)
        elif a.worker:
            if a.remaining is None or not 0<a.remaining<=240:raise p.Stop('A bounded remaining allowance is required.')
            worker(a.root,a.round,a.remaining,a.pause_after)
        elif a.run:
            r=run(a.root,a.round,a.resume,a.pause_after);print('RESULT:',r['status']);print('REPORT:',report_path(a.root,a.round))
        else:
            r=p.read_json(report_path(a.root,a.round));export(a.root,a.round,r,charts(r))
    except BaseException as e:
        print('RESULT: STOPPED',flush=True);print('ERROR:',type(e).__name__+': '+str(e),flush=True);raise

if __name__=='__main__':main()
