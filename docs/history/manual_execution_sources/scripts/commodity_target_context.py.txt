def metadata_map(pairs):
    import pandas as pd
    if (not isinstance(pairs,pd.DataFrame) or pairs.empty or not pairs.columns.is_unique
        or not {'target','pair','lag'}.issubset(pairs.columns)
        or pairs[['target','pair','lag']].isna().any().any()
        or pairs.target.duplicated().any() or pd.api.types.is_bool_dtype(pairs.lag)
        or not pairs.lag.isin([1,2,3,4]).all()
        or not pairs.target.map(lambda s:isinstance(s,str) and bool(s)).all()):
        raise Stop('Invalid ordered target/pair/horizon metadata.')
    legs=[]
    for value in pairs.pair:
        if not isinstance(value,str):
            raise Stop('Pair identifiers must be strings.')
        row=value.split(' - ')
        if len(row) not in (1,2) or len(set(row))!=len(row) or any(not a or a!=a.strip() for a in row):
            raise Stop('Invalid or repeated target legs.')
        legs.append(row)
    counts=Counter(a for row in legs for a in row)
    anchors=sorted(counts,key=lambda a:(-counts[a],a))[:MAX_ANCHORS]
    return legs,anchors,counts

def build_context(x,pairs):
    """Metadata and raw market observations only; no outcome argument exists."""
    import numpy as np
    import pandas as pd
    legs,anchors,counts=metadata_map(pairs)
    if (not isinstance(x,pd.DataFrame) or x.empty or len(x)>STOP or not x.columns.is_unique
        or not pd.api.types.is_integer_dtype(x.index) or x.index.tolist()!=list(range(len(x)))
        or any(a not in x.columns for row in legs for a in row)
        or not all(pd.api.types.is_numeric_dtype(x[a]) and not pd.api.types.is_bool_dtype(x[a]) for a in anchors)):
        raise Stop('Bounded, numeric, contiguous market inputs and all target assets are required.')
    signed=np.array([[int(row[0]==a)-int(len(row)==2 and row[1]==a) for a in anchors]
                     for row in legs],dtype=np.float32)
    present=np.abs(signed)
    raw=x[anchors].to_numpy(dtype=float,copy=True)
    valid=np.isfinite(raw)&(raw>0)
    logs=np.full_like(raw,np.nan)
    np.log(raw,out=logs,where=valid)
    returns=pd.DataFrame(logs,index=x.index,columns=anchors).diff()
    sigma=returns.shift(1).rolling(WINDOW,min_periods=MIN_HISTORY).std(ddof=0).clip(lower=1e-6)
    shocks=(returns/sigma).clip(-12,12).to_numpy(dtype=float,copy=True)
    arrays=[]; names=[]; kinds=[]
    for j,a in enumerate(anchors):
        for label,weights in [('signed',signed[:,j]),('present',present[:,j])]:
            arrays.append(np.broadcast_to(weights,(len(x),len(pairs))))
            names.append(f'target_context__identity_{label}__{a}'); kinds.append('identity')
    for j,a in enumerate(anchors):
        for label,weights in [('signed',signed[:,j]),('present',present[:,j])]:
            arrays.append(np.where(weights[None,:]!=0,shocks[:,j,None]*weights[None,:],0))
            names.append(f'target_context__shock_{label}__{a}'); kinds.append('shock')
    block=np.stack(arrays,axis=2).astype(np.float32)
    if np.isinf(block).any() or len(names)!=len(set(names)):
        raise Stop('Invalid target-context features.')
    membership=present.sum(axis=1)
    metadata=dict(anchors=anchors,anchor_counts=[int(counts[a]) for a in anchors],
        unique_underlyings=len(counts),targets=len(pairs),templates=len(names),
        identity_templates=2*len(anchors),shock_templates=2*len(anchors),
        targets_with_anchor=int((membership>0).sum()),targets_without_anchor=int((membership==0).sum()),
        fully_anchored_targets=sum(all(a in anchors for a in row) for row in legs),
        names=names,kinds=kinds,
        selection='Supplied target metadata only; no observed-label counts or outcome values.')
    return block,names,metadata

def variant_names(names,variant):
    if variant not in VARIANTS or len(names)!=len(set(names)):
        raise Stop('Unknown panel or duplicate column names.')
    identity=[n for n in names if n.startswith('target_context__identity_')]
    shocks=[n for n in names if n.startswith('target_context__shock_')]
    if not identity or len(identity)!=len(shocks) or identity+shocks!=names:
        raise Stop('Incomplete identity/shock feature definition.')
    return identity if variant=='context_identity' else shocks if variant=='context_shocks' else names

def append_panel(base,block,names,variant):
    import numpy as np
    from commodity_prediction.domain.catalog import Panel
    chosen=variant_names(names,variant)
    if block.shape != (len(base.dates),len(base.targets),len(names)) or set(names)&set(base.names):
        raise Stop('New context array does not align with the frozen panel.')
    indices=[names.index(n) for n in chosen]
    panel=Panel(np.concatenate([base.values,block[:,:,indices]],axis=2),
                base.dates,base.targets,base.names+chosen,dict(base.source_series))
    panel.validate()
    return panel

def smoke():
    import numpy as np
    import pandas as pd
    rng=np.random.default_rng(20260912)
    x=pd.DataFrame(np.exp(np.cumsum(rng.normal(0,.015,(100,3)),axis=0)),columns=['A','B','C'])
    p=pd.DataFrame({'target':['target_0','target_1','target_2'],
                    'pair':['A - B','B - A','C'],'lag':[1,2,4]})
    before=x.copy(deep=True)
    a,n,m=build_context(x,p)
    b,_,_=build_context(x.iloc[:77],p)
    np.testing.assert_array_equal(a[:77],b)
    changed=x.copy(deep=True);changed.iloc[80:]*=2
    c,_,_=build_context(changed,p)
    np.testing.assert_array_equal(a[:80],c[:80])
    pd.testing.assert_frame_equal(x,before,check_exact=True)
    bad=x.copy(deep=True);bad.loc[80,'A']=np.inf
    broken,bnames,_=build_context(bad,p)
    j=bnames.index('target_context__shock_signed__A')
    if not np.isnan(broken[80,0,j]) or broken[80,2,j]!=0:
        raise Stop('Invalid member quote or structural nonmember encoding is incorrect.')
    return dict(status='TARGET_CONTEXT_TIMING_SMOKE_PASSED',future_perturbation_exact=True,
                prefix_exact=True,input_unchanged=True,pandas_version=pd.__version__,numpy_version=np.__version__)

def audit_features(block,names,meta,base,pairs):
    import numpy as np
    def signature(a):
        a=np.array(a,dtype=np.float32,copy=True,order='C');a[np.isnan(a)]=np.nan
        return hashlib.sha256(a.tobytes()).hexdigest()
    history={signature(base.values[252:TRAIN_STOP,:,i]):n for i,n in enumerate(base.names)}
    legs,anchors,_=metadata_map(pairs)
    rows=[]
    for i,name in enumerate(names):
        values=block[252:TRAIN_STOP,:,i]
        anchor=anchors[(i%(2*len(anchors)))//2]
        members=np.array([anchor in row for row in legs])
        z=values[:,members]; finite=z[np.isfinite(z)]
        sig=signature(values); duplicate=history.get(sig)
        if duplicate is None: history[sig]=name
        rows.append(dict(name=name,kind=meta['kinds'][i],anchor=anchor,
                         applicable_targets=int(members.sum()),
                         finite_fraction_all=float(np.isfinite(values).mean()),
                         finite_fraction_members=float(np.isfinite(z).mean()),
                         q05_members=float(np.quantile(finite,.05)) if len(finite) else None,
                         q95_members=float(np.quantile(finite,.95)) if len(finite) else None,
                         exact_duplicate_of=duplicate,training_sha256=sig))
    return rows

def fit_one(root,u,name,stage,lineage,base,block,names,y,pairs,fold,config,report):
    import joblib
    import pandas as pd
    from commodity_prediction.domain.model import prepare,select
    from commodity_prediction.domain.attribution.run import fit_stage
    from commodity_prediction.domain.experiment import Experiment
    if name not in VARIANTS or stage.exists():
        raise Stop('Undeclared/existing context stage; no automatic refit.')
    panel=append_panel(base,block,names,name)
    settings={**config,'max_features':len(panel.names),'max_abs_correlation':1.01}
    stats=prepare(panel,y,fold.train_stop,settings)
    experiment=Experiment(name,algorithm='histogram')
    _,audit=select(stats,experiment.candidates(panel.names),settings,False)
    admitted=[n for n in audit['selected_names'] if n.startswith('target_context__')]
    numeric=[n for n in admitted if n.startswith('target_context__shock_')]
    if not admitted or (name!='context_identity' and not numeric):
        raise Stop('No proposed usable inputs survived training-only checks; no fit started.')
    if set(audit['rejection_reasons'])&{'feature_budget','correlated','unstable_sign'}:
        raise Stop('Undeclared screening exclusion.')
    if report['fit_attempts_this_call']>=3:
        raise Stop('Three-fit limit exhausted.')
    report['fit_attempts_this_call']+=1;u.atomic_json(report_path(root),report)
    result=fit_stage(panel,y,pairs,fold,experiment,settings,stats,stage,lineage)
    saved=pd.read_parquet(stage/'predictions.parquet')
    replay=joblib.load(stage/'model.joblib').predict(panel,START,STOP)
    error=u.exact_predictions(saved,replay)
    recorded={**result,'context_inputs_admitted':len(admitted),'shock_inputs_admitted':len(numeric)}
    report['results'].append(recorded);report['new_training_fits']+=1
    report['maximum_prediction_replay_error']=max(error,report['maximum_prediction_replay_error'])
    report['checkpoint_hashes'][name]=u.verify_stage(stage,lineage)
    u.atomic_json(report_path(root),report)
    emit('CONTEXT_MODEL_VERIFIED',variant=name,completed=report['new_training_fits'],total=3,
         fold=FOLD,official_metric=result['metrics']['official_metric'])
    del panel,stats,replay;gc.collect()

def decisions(results,control,metric):
    if len(results)!=3 or {r['variant'] for r in results}!=set(VARIANTS):
        raise Stop('Incomplete context result inventory.')
    baseline=metric(control['daily_rank_correlations'],180)
    if not math.isclose(baseline,control['official_metric'],rel_tol=0,abs_tol=1e-12):
        raise Stop('Control metric differs from its daily scores.')
    scores={r['variant']:metric(r['metrics']['daily_rank_correlations'],180) for r in results}
    rows=[];eligible=[]
    for r in results:
        name=r['variant'];score=scores[name]
        if r.get('fold')!=FOLD or r.get('train_stop')!=TRAIN_STOP or r.get('validation_start')!=START or r.get('validation_stop')!=STOP:
            raise Stop('Fitted result belongs to a different training/validation period.')
        if not math.isclose(score,r['metrics']['official_metric'],rel_tol=0,abs_tol=1e-12):
            raise Stop('Result metric differs from daily scores.')
        selected=r['selection']['selected_names']
        admitted=sum(n.startswith('target_context__') for n in selected)
        numeric=sum(n.startswith('target_context__shock_') for n in selected)
        if admitted!=r['context_inputs_admitted'] or numeric!=r['shock_inputs_admitted'] or not admitted:
            raise Stop('Input-admission evidence differs.')
        delta=score-baseline; d_identity=score-scores['context_identity']
        gate=delta>=GAIN and (name=='context_identity' or numeric>0)
        if name=='context_joint': gate=gate and d_identity>=GAIN
        row=dict(variant=name,official_metric=score,delta_vs_current_market=delta,
                 delta_vs_identity=d_identity,context_inputs_admitted=admitted,
                 shock_inputs_admitted=numeric,passes_review_gate=bool(gate))
        rows.append(row)
        if gate: eligible.append(name)
    return rows,eligible

        r.update(status='TARGET_CONTEXT_REVIEW_READY',source_unchanged=True,rank_parent_unchanged=True,
            finished_utc=utc(),limitations=[
            'The middle fold was chosen after observing historical weakness, before this experiment. It is not a fresh holdout.',
            'Five contrasts condition on fitted models; no full adaptive-search multiplicity correction.',
            'Membership adds entity identity; it is not external fundamental data or proof of causal exposure.',
            'Top 16 are chosen from target metadata frequency, not importance. Omitted underlyings and applicability are reported.',
            'Static identity and return interactions can be redundant with existing pooled representations; exact active-panel duplicates are audited.',
            'Training on a fixed universe is not evidence of generalization to unseen instruments.',
            'Numerical split changes depend on the fixed learner. A negative result does not exhaust raw-input or learned-representation approaches.',
            'Private ZIP remains on the same disk until downloaded; no cloud/Git writes or automatic shutdown.'])
        save();u.atomic_json(directory/'review.json',r);return r
    except BaseException as exc:
        r.update(status='STOPPED',error=type(exc).__name__+': '+str(exc),finished_utc=utc());save();raise
    finally:
        signal.setitimer(signal.ITIMER_REAL,0);signal.signal(signal.SIGALRM,handler)

def verify_complete(root,r,session,u):
    lineage,evidence=identity(root,u);read_prior(root,u)
    if (r.get('status') not in TERMINAL or r.get('lineage')!=lineage or r.get('identity')!=evidence
        or r.get('new_training_fits')!=3 or r.get('fit_attempts_this_call')!=3
        or r.get('control_refits')!=0 or r.get('maximum_prediction_replay_error')!=0
        or r.get('final_test_evaluations')!=0 or r.get('parents_unchanged') is not True
        or r.get('source_unchanged') is not True or r.get('rank_parent_unchanged') is not True
        or r.get('fold')!=FOLD or set(r.get('checkpoint_hashes',{}))!=set(VARIANTS)
        or not session.close(r.get('control',{}).get('official_metric',float('nan')),CONTROL)
        or r.get('control',{}).get('date_ids')!=list(range(START,STOP))):
        raise Stop('Not a complete verified target-context study.')
    rows,eligible=decisions(r['results'],r['control'],session.metric)
    decision='REVIEW_TARGET_CONTEXT_ON_OTHER_PERIODS' if eligible else 'STOP_TESTED_TARGET_CONTEXT_PANELS'
    if rows!=r['comparison_rows'] or eligible!=r['candidates_for_review'] or decision!=r['decision']:
        raise Stop('Stored decision differs from the declared computation.')
    directory=root/'artifacts/target_context'/lineage
    u.verify_stage(directory/'features',lineage,r['feature_checkpoint_hashes'])
    for name,pins in r['checkpoint_hashes'].items():
        stage=directory/'fold_1'/name;u.verify_stage(stage,lineage,pins)
        record=next(v for v in r['results'] if v['variant']==name)
        if u.read_json(stage/'result.json')!={k:v for k,v in record.items() if k not in ['context_inputs_admitted','shock_inputs_admitted']}:
            raise Stop('Report differs from sealed fitted result.')

def run(root=ROOT):
    root=Path(root);session,older,u=previous(root)
    ready=u.readiness(root);u.validate_runtime(root,ready);u.validate_source(root);read_prior(root,u)
    out=u.safe_path(root,'logs/manual_target_context');out.mkdir(parents=True,exist_ok=True)
    with u.safe_path(out,'supervisor.lock').open('a') as lock:
        fcntl.flock(lock.fileno(),fcntl.LOCK_EX|fcntl.LOCK_NB)
        if report_path(root).exists():
            r=u.read_json(report_path(root))
            if r.get('status') not in TERMINAL:
                raise Stop('Interrupted context run exists. Preserve its report; no automatic retry or budget reset.')
            verify_complete(root,r,session,u);emit('COMPLETED_CONTEXT_REUSED',new_fits=0)
            return {**r,'new_fits_this_notebook_call':0}
        lineage,_=identity(root,u)
        if (root/'artifacts/target_context'/lineage).exists(): raise Stop('Orphan context directory preserved; no refit.')
        if shutil.disk_usage(root).free<1024**3: raise Stop('Need 1 GiB free; do not delete old projects blindly.')
        u.atomic_json(report_path(root),dict(status='STARTING_WORKER',lineage=lineage,new_training_fits=0,started_utc=utc()))
        args=[ready['runtime']['python'],'-u',str(Path(__file__).resolve()),'--worker','--root',str(root)]
        began=time.monotonic()
        try:
            session._supervise_command(args,root,LIMIT,out/'target_context_worker.log',u.environment(root))
        except BaseException as exc:
            r=u.read_json(report_path(root))
            if r.get('status')=='STARTING_WORKER': r['new_training_fits']=None
            r.update(status='STOPPED',supervisor_error=type(exc).__name__+': '+str(exc),
                     supervised_seconds=round(time.monotonic()-began,3))
            u.atomic_json(report_path(root),r);raise
        r=u.read_json(report_path(root));verify_complete(root,r,session,u)
        r.update(supervised_seconds=round(time.monotonic()-began,3),new_fits_this_notebook_call=3)
        u.atomic_json(report_path(root),r);return r

def private_backup(root,r,u):
    base=root/'artifacts/target_context'/r['lineage']
    paths=[base/'plan.json',base/'review.json']+[base/'features'/n for n in r['feature_checkpoint_hashes']]
    for name in VARIANTS: paths += [base/'fold_1'/name/n for n in r['checkpoint_hashes'][name]]
    pins={str(p.relative_to(root)):u.digest(u.safe_path(root,str(p.relative_to(root)))) for p in paths}
    if sum(p.stat().st_size for p in paths)>256*1024**2: raise Stop('Backup exceeds 256 MiB; fitted stages preserved.')
    dest=u.safe_path(output_dir(root),'target_context_checkpoints.zip');tmp=u.safe_path(dest.parent,dest.name+'.tmp')
    with zipfile.ZipFile(tmp,'w',zipfile.ZIP_DEFLATED,compresslevel=1) as z:
        for p in paths:z.write(p,str(p.relative_to(root)))
        z.writestr('SHA256SUMS.json',json.dumps(pins,sort_keys=True,indent=2))
    with zipfile.ZipFile(tmp) as z:
        if z.testzip():raise Stop('Backup CRC verification failed.')
        for name,h in pins.items():
            if hashlib.sha256(z.read(name)).hexdigest()!=h:raise Stop('Backup checksum differs.')
    os.replace(tmp,dest)
    return dict(path=str(dest),sha256=u.digest(dest),bytes=dest.stat().st_size,files=len(paths),
                private=True,contains_features_models_predictions=True,off_disk_backup_confirmed=False,
                raw_csvs_included=False)
