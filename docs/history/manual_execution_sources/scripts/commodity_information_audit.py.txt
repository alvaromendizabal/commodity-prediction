def ordinal(a):
    """Row ranks among finite predictors, never fill missing observations."""
    import numpy as np
    import pandas as pd
    x = np.array(a, dtype=float, copy=True)
    if x.ndim != 2:
        raise Stop('Rank input must have two axes.')
    x[~np.isfinite(x)] = np.nan
    finite = np.isfinite(x)
    count = finite.sum(axis=1)
    ranked = pd.DataFrame(x).rank(axis=1, method='average').to_numpy(dtype=float, copy=True)
    result = 2 * (ranked - 1) / np.maximum(count[:, None] - 1, 1) - 1
    result[count < 3] = np.nan
    return result

def horizon_ranks(values, horizons):
    import numpy as np
    out = np.full_like(values, np.nan, dtype=float)
    for h in (1,2,3,4):
        idx = np.flatnonzero(horizons == h)
        if len(idx):
            out[:,idx] = ordinal(values[:,idx])
    return out


def build_relative(values, names, pairs):
    """Transform supplied training panel only; no outcome argument or future fit."""
    import numpy as np
    import pandas as pd
    a = check_inputs(values, names, pairs)
    horizons = pairs.lag.to_numpy(dtype=int, copy=True)
    arrays, output_names, metadata = [], [], []
    for source in SIGNALS:
        raw = np.array(a[:,:,names.index(source)], dtype=float, copy=True)
        history = pd.DataFrame(raw).shift(1).rolling(63, min_periods=42)
        mean = history.mean().to_numpy(dtype=float, copy=True)
        sd = history.std(ddof=0).to_numpy(dtype=float, copy=True)
        surprise = np.divide(raw-mean, sd, out=np.full_like(raw,np.nan), where=sd>1e-12)
        channels = [ordinal(raw), horizon_ranks(raw,horizons), horizon_ranks(surprise,horizons)]
        for kind, channel in zip(KINDS,channels,strict=True):
            name = 'relative_state__' + source.replace('__','_') + '__' + kind
            arrays.append(channel)
            output_names.append(name)
            metadata.append(dict(name=name, source=source, kind=kind,
                source_uses_released_labels=source.startswith('released_priors__')))
    cube = np.stack(arrays,axis=2).astype(np.float32)
    if np.isinf(cube).any():
        raise Stop('Relative features contain infinity.')
    return cube, output_names, metadata


def daily_ic(a, y):
    """Pairwise-observed daily Spearman; not pooled or in-sample model importance."""
    import numpy as np
    import pandas as pd
    x = np.asarray(a,dtype=float)
    truth = np.asarray(y,dtype=float)
    if x.ndim != 2 or x.shape != truth.shape:
        raise Stop('Daily association axes differ.')
    valid = np.isfinite(x) & np.isfinite(truth) & (truth != -999999)
    xr = pd.DataFrame(np.where(valid,x,np.nan)).rank(axis=1).to_numpy(dtype=float,copy=True)
    yr = pd.DataFrame(np.where(valid,truth,np.nan)).rank(axis=1).to_numpy(dtype=float,copy=True)
    n = valid.sum(axis=1)
    xm = np.where(valid,xr,0).sum(axis=1) / np.maximum(n,1)
    ym = np.where(valid,yr,0).sum(axis=1) / np.maximum(n,1)
    dx = np.where(valid,xr-xm[:,None],0)
    dy = np.where(valid,yr-ym[:,None],0)
    denom = np.sqrt((dx*dx).sum(axis=1)*(dy*dy).sum(axis=1))
    return np.divide((dx*dy).sum(axis=1),denom,out=np.full(len(x),np.nan),where=(n>=3)&(denom>1e-12))


def signature(a):
    import numpy as np
    x = np.array(a,dtype=np.float32,copy=True,order='C')
    x[np.isnan(x)] = np.nan
    x[x == 0] = 0  # canonicalize negative zero for the duplicate audit
    return hashlib.sha256(x.tobytes()).hexdigest()


def candidate_screen(base, base_names, cube, meta, y):
    import numpy as np
    if len(base)!=TRAIN_STOP or len(cube)!=TRAIN_STOP or np.asarray(y).shape!=cube.shape[:2]:
        raise Stop('Screen must use exactly the first 1164 training rows.')
    lo, mid, end = 252, 708, TRAIN_STOP
    existing = {signature(base[lo:end,:,i]):n for i,n in enumerate(base_names)}
    rows = []
    for i,item in enumerate(meta):
        data = cube[lo:end,:,i]
        digest = signature(data)
        duplicate = existing.get(digest)
        if duplicate is None:
            existing[digest] = item['name']
        ic = daily_ic(cube[:,:,i],y)
        first, second = ic[lo:mid], ic[mid:end]
        left, right = first[np.isfinite(first)], second[np.isfinite(second)]
        m1 = float(left.mean()) if len(left) else None
        m2 = float(right.mean()) if len(right) else None
        coverage = float(np.isfinite(data).mean())
        consistent = m1 is not None and m2 is not None and m1*m2 > 0
        eligible = (item['kind']=='innovation_horizon_rank' and duplicate is None and coverage>=.70
                    and len(left)>=40 and len(right)>=40 and consistent)
        raw_ic = daily_ic(base[:,:,base_names.index(item['source'])], y)
        valid = np.isfinite(raw_ic[lo:end]) & np.isfinite(ic[lo:end])
        error = float(np.max(np.abs(raw_ic[lo:end][valid]-ic[lo:end][valid]))) if valid.any() else None
        if item['kind']=='global_rank':
            if (not np.array_equal(np.isfinite(raw_ic[lo:end]),np.isfinite(ic[lo:end]))
                or (error is not None and error>1e-12)):
                raise Stop('Expected monotonic global-rank IC invariance failed.')
        rows.append(dict(**item,training_coverage=coverage,half_1_mean_ic=m1,half_2_mean_ic=m2,
            half_1_eligible_dates=len(left),half_2_eligible_dates=len(right),same_direction=bool(consistent),
            exact_duplicate_of=duplicate,training_sha256=digest,eligible_innovation=bool(eligible),
            training_relevance=min(abs(m1),abs(m2)) if eligible else 0.0,
            max_daily_ic_change_vs_source=error,
            selection_note='Screened innovation' if item['kind']=='innovation_horizon_rank' else 'Representation control; not selected as novel signal by marginal IC'))
    selected = sorted([r for r in rows if r['eligible_innovation']], key=lambda r:(-r['training_relevance'],r['name']))[:6]
    return dict(rows=rows,shortlist=[r['name'] for r in selected],training_rows=TRAIN_STOP,
                screen_start=lo,screen_stop_exclusive=end,split_date=mid,
                validation_rows_used_for_candidate_selection=0,
                decision='REVIEW_INNOVATION_CANDIDATES_BEFORE_FITS' if selected else 'NO_INNOVATION_PASSED_THIS_SCREEN')


def block_order(length, block, seed):
    import numpy as np
    if length < 2*block or block < 1:
        raise Stop('Need at least two nonempty diagnostic blocks.')
    chunks = [np.arange(s,min(s+block,length)) for s in range(0,length,block)]
    order = np.random.default_rng(seed).permutation(len(chunks))
    if np.array_equal(order,np.arange(len(chunks))):
        order = np.roll(order,1)
    return np.concatenate([chunks[i] for i in order])


def permuted_values(values, names, family, order):
    import numpy as np
    a = np.asarray(values)
    order = np.asarray(order)
    if a.ndim!=3 or a.shape[2]!=len(names) or len(set(names))!=len(names):
        raise Stop('Permutation panel axes differ.')
    if order.dtype.kind not in 'iu' or not np.array_equal(np.sort(order),np.arange(len(a))):
        raise Stop('Diagnostic order must permute each row exactly once.')
    idx = [i for i,n in enumerate(names) if n.split('__',1)[0]==family]
    if not idx:
        raise Stop('Declared family not in the panel.')
    out = np.array(a,copy=True)
    out[:,:,idx] = a[order][:,:,idx]
    return out


def rank_summary(daily):
    import numpy as np
    v = np.asarray(daily,float)
    if v.ndim!=1 or len(v)<2 or not np.isfinite(v).all() or (abs(v)>1+1e-12).any() or v.std(ddof=0)<=1e-12:
        raise Stop('Undefined diagnostic correlation metric.')
    return dict(official_metric=float(v.mean()/v.std(ddof=0)),mean_daily_rank_correlation=float(v.mean()),
                std_daily_rank_correlation=float(v.std(ddof=0)),daily_rank_correlations=v.tolist())


def summarize_reliance(folds):
    import numpy as np
    if len(folds)!=3 or [r['fold'] for r in folds]!=[0,1,2]:
        raise Stop('Three chronological frozen-control audits required.')
    full = np.concatenate([q['control']['daily_rank_correlations'] for q in folds])
    mean = np.concatenate([q['mean_only']['daily_rank_correlations'] for q in folds])
    if len(full)!=535:
        raise Stop('Historical evaluation dates differ.')
    control, intercept = rank_summary(full), rank_summary(mean)
    rows = []
    for family in FAMILIES:
        drops=[]
        for seed in SEEDS:
            parts=[]
            for fold in folds:
                matches=[r for r in fold['permutations'] if r['family']==family and r['seed']==seed]
                if len(matches)!=1:
                    raise Stop('Incomplete family/repetition diagnostics.')
                parts.append(matches[0]['daily_rank_correlations'])
            score=rank_summary(np.concatenate(parts))['official_metric']
            drops.append(control['official_metric']-score)
        rows.append(dict(family=family,pooled_metric_drop_repetitions=drops,
                         mean_drop=float(np.mean(drops)),min_drop=float(min(drops)),max_drop=float(max(drops))))
    return dict(control=control,mean_only=intercept,metric_difference_vs_mean=control['official_metric']-intercept['official_metric'],
                rows=rows,scope='Frozen-model diagnostic, not a new trained system; two perturbations are not uncertainty confidence bounds.')

def audit_fold(base, y, pairs, fold, model, saved, u, metric_fn):
    import numpy as np
    import pandas as pd
    from commodity_prediction.domain.catalog import Panel
    number,train_stop,start,stop=fold
    if saved.index.tolist()!=list(range(start,stop)) or saved.columns.tolist()!=base.targets:
        raise Stop('Saved prediction dates or targets differ.')
    replay=model.predict(base,start,stop)
    error=u.exact_predictions(saved,replay)
    control=metric_fn(y.loc[saved.index],saved,pairs)
    if not math.isclose(control['official_metric'],SCORES[number],abs_tol=1e-12,rel_tol=0):
        raise Stop('Frozen control did not reproduce the recorded score.')
    # Read the fitted means; do not estimate another model or optimize residual weights.
    target_mean=np.asarray(model.target_mean,dtype=float)
    if target_mean.shape!=(len(base.targets),) or not np.isfinite(target_mean).all():
        raise Stop('Invalid saved target-mean vector.')
    predicted_mean=pd.DataFrame(np.broadcast_to(target_mean,saved.shape).copy(),index=saved.index,columns=saved.columns)
    mean_only=metric_fn(y.loc[saved.index],predicted_mean,pairs)
    permutations=[]
    for seed in SEEDS:
        order=block_order(stop-start,20,seed)
        for family in FAMILIES:
            changed=permuted_values(base.values[start:stop],base.names,family,order)
            perturb=Panel(changed,base.dates[start:stop],base.targets,base.names,dict(base.source_series))
            pred=model.predict(perturb,0,stop-start)
            metrics=metric_fn(y.loc[pred.index],pred,pairs)
            permutations.append(dict(family=family,seed=seed,
                selected_family_templates=sum(n.split('__',1)[0]==family for n in model.feature_names),
                official_metric=metrics['official_metric'],metric_drop=control['official_metric']-metrics['official_metric'],
                daily_rank_correlations=metrics['daily_rank_correlations']))
            emit('FAMILY_DIAGNOSTIC_COMPLETE',fold=number,family=family,seed=seed,new_fits=0)
            del changed,perturb,pred
    return dict(fold=number,train_stop=train_stop,validation_start=start,validation_stop=stop,
                control=control,mean_only=mean_only,maximum_prediction_replay_error=error,permutations=permutations)

def run(root=ROOT):
    root=Path(root);session,older,u=previous(root)
    ready=u.readiness(root);u.validate_runtime(root,ready);u.validate_source(root);read_prior(root,u)
    out=u.safe_path(root,'logs/manual_information_audit');out.mkdir(parents=True,exist_ok=True)
    with u.safe_path(out,'supervisor.lock').open('a') as lock:
        fcntl.flock(lock.fileno(),fcntl.LOCK_EX|fcntl.LOCK_NB)
        if report_path(root).exists():
            r=u.read_json(report_path(root))
            if r.get('status') not in TERMINAL:
                raise Stop('Interrupted audit retained; do not reset its time budget or repeat unchanged.')
            verify_complete(root,r,u);emit('COMPLETED_AUDIT_REUSED',new_fits=0)
            return r
        lineage,_=identity(root,u)
        if (root/'artifacts/information_audit'/lineage).exists():raise Stop('Orphan audit checkpoint retained.')
        if shutil.disk_usage(root).free<1024**3:raise Stop('Need 1 GiB free; keep old data/environment directories.')
        u.atomic_json(report_path(root),dict(status='STARTING_WORKER',lineage=lineage,new_training_fits=0,started_utc=utc()))
        cmd=[ready['runtime']['python'],'-u',str(Path(__file__).resolve()),'--worker','--root',str(root)]
        began=time.monotonic()
        try:
            session._supervise_command(cmd,root,LIMIT,out/'information_audit_worker.log',u.environment(root))
        except BaseException as exc:
            r=u.read_json(report_path(root));r.update(status='STOPPED',supervisor_error=type(exc).__name__+': '+str(exc),
                supervised_seconds=round(time.monotonic()-began,3));u.atomic_json(report_path(root),r);raise
        r=u.read_json(report_path(root));verify_complete(root,r,u)
        r['supervised_seconds']=round(time.monotonic()-began,3);u.atomic_json(report_path(root),r)
        return r
