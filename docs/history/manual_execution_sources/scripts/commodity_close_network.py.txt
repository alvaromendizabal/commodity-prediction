def pairwise_correlation(follower, leader):
    """Pearson correlations using pairwise observed rows, no filling by future means."""
    import numpy as np
    a, b = np.asarray(follower, dtype=float), np.asarray(leader, dtype=float)
    if a.ndim != 2 or b.ndim != 2 or a.shape[0] != b.shape[0] or a.shape[0] < MIN_OBS:
        raise Stop('Correlation history axes or length invalid.')
    f, g = np.isfinite(a).astype(float), np.isfinite(b).astype(float)
    aa, bb = np.where(np.isfinite(a), a, 0.0), np.where(np.isfinite(b), b, 0.0)
    count = f.T @ g
    sx, sy = aa.T @ g, f.T @ bb
    n = np.maximum(count, 1)
    cov = aa.T @ bb - sx * sy / n
    vx = np.maximum((aa * aa).T @ g - sx * sx / n, 0)
    vy = np.maximum(f.T @ (bb * bb) - sy * sy / n, 0)
    den = np.sqrt(vx * vy)
    good = (count >= MIN_OBS) & (vx > 1e-20) & (vy > 1e-20)
    corr = np.divide(cov, den, out=np.full_like(cov, np.nan), where=good)
    return np.clip(corr, -1, 1), count


def neighbors(corr, counts, left, right):
    """Select incoming peers independently of labels and excluding both target legs."""
    import numpy as np
    m, n = len(left), corr.shape[0]
    if corr.shape != (n, n) or counts.shape != corr.shape or len(right) != m:
        raise Stop('Network axes mismatch.')
    if any(i < 0 or i >= n for i in left) or any(j < -1 or j >= n for j in right):
        raise Stop('Invalid target-to-asset mapping.')
    index = np.zeros((m, 2, K), dtype=int)
    weights = np.zeros((m, 2, K), dtype=float)
    for target, (a, b) in enumerate(zip(left, right, strict=True)):
        for leg, anchor in enumerate((a, b)):
            if anchor < 0:
                continue
            eligible = np.isfinite(corr[anchor]) & (counts[anchor] >= MIN_OBS) & (np.abs(corr[anchor]) >= MIN_CORR)
            eligible[a] = False
            if b >= 0:
                eligible[b] = False
            ids = np.flatnonzero(eligible)
            # Stable tie break uses the canonically sorted asset index.
            ids = sorted(ids, key=lambda j: (-abs(float(corr[anchor, j])), int(j)))[:K]
            if len(ids) < MIN_PEERS:
                continue
            index[target, leg, :len(ids)] = ids
            weights[target, leg, :len(ids)] = corr[anchor, ids]
    return index, weights


def aggregate_peers(values, index, weights):
    """Signed correlation-weighted mean; weak observed coverage remains missing."""
    import numpy as np
    a = np.asarray(values, dtype=float)
    sampled = a[:, index]  # rows, target, leg, peer
    active = np.abs(weights) > 0
    finite = np.isfinite(sampled) & active[None]
    observed_mass = (np.abs(weights)[None] * finite).sum(axis=-1)
    total = np.abs(weights).sum(axis=-1)
    fraction = np.divide(observed_mass, total[None], out=np.zeros_like(observed_mass), where=total[None] > 0)
    good = (finite.sum(axis=-1) >= MIN_PEERS) & (fraction >= MIN_WEIGHT_FRACTION)
    numerator = (np.where(finite, sampled, 0) * weights[None]).sum(axis=-1)
    result = np.divide(numerator, observed_mass, out=np.full_like(numerator, np.nan), where=good)
    return result, fraction


def validate_inputs(x, pairs):
    import numpy as np
    import pandas as pd
    if not isinstance(x, pd.DataFrame) or not isinstance(pairs, pd.DataFrame):
        raise Stop('Expected market and target DataFrames.')
    if (not x.columns.is_unique or not pairs.columns.is_unique or pairs.empty
            or not {'target', 'pair', 'lag'}.issubset(pairs.columns)):
        raise Stop('Ambiguous market columns or incomplete target metadata.')
    dates = x.index.to_numpy()
    if (len(x) < WINDOW + 2 or not np.issubdtype(dates.dtype, np.integer)
            or dates.tolist() != list(range(len(x))) or len(x) > VALIDATION_STOP):
        raise Stop('Use the contiguous zero-based first-fold prefix only.')
    if (pairs[['target', 'pair', 'lag']].isna().any().any() or pairs.target.duplicated().any()
            or not pairs.lag.isin([1, 2, 3, 4]).all()):
        raise Stop('Invalid target/horizon contract.')
    legs = [v.split(' - ') if isinstance(v, str) else [] for v in pairs.pair]
    if any(len(v) not in (1, 2) or len(set(v)) != len(v) or any(not a for a in v) for v in legs):
        raise Stop('Invalid target leg identifiers.')
    assets = sorted({a for row in legs for a in row})
    if len(assets) < MIN_PEERS + 2 or not set(assets).issubset(x.columns):
        raise Stop('Insufficient or missing target-asset price columns.')
    if not all(pd.api.types.is_numeric_dtype(x[a]) for a in assets):
        raise Stop('Target-asset prices must be numeric.')
    return assets, legs


def network_features(x, pairs, progress=False):
    """Build 16 causal templates; no target outcomes accepted by this function.

    The directed matrix uses follower r_s and peer r_(s-1), s < refresh t.
    Same-time and directed matrices have identical windows/top-K/standardization.
    Coefficients and scales remain fixed for 21 rows, then use a new past-only window.
    """
    import numpy as np
    import pandas as pd
    assets, legs = validate_inputs(x, pairs)
    loc = {a: i for i, a in enumerate(assets)}
    left = np.array([loc[v[0]] for v in legs])
    right = np.array([loc[v[1]] if len(v) == 2 else -1 for v in legs])
    raw = x[assets].to_numpy(dtype=float)
    logs = np.log(np.where(np.isfinite(raw) & (raw > 0), raw, np.nan))
    returns = np.vstack([np.full((1, len(assets)), np.nan), np.diff(logs, axis=0)])
    n, m = len(x), len(pairs)
    names = MASKS + NUMERIC['synchronous'] + NUMERIC['directed']
    values = np.full((n, m, len(names)), np.nan, dtype=np.float32)
    values[:, :, :len(MASKS)] = 0
    refreshes, snapshots = [], []
    for start in range(WINDOW + 1, n, REFRESH):
        stop = min(n, start + REFRESH)
        follower = returns[start-WINDOW:start]
        count = np.isfinite(follower).sum(axis=0)
        total = np.where(np.isfinite(follower), follower, 0).sum(axis=0)
        mean = np.divide(total, count, out=np.full(len(assets), np.nan), where=count >= MIN_OBS)
        dif = np.where(np.isfinite(follower), follower - mean, 0)
        variance = np.divide((dif * dif).sum(axis=0), count, out=np.full(len(assets), np.nan), where=count >= MIN_OBS)
        scale = np.sqrt(variance)
        scale[~np.isfinite(scale) | (scale <= 1e-10)] = np.nan
        # Include only the small lag prefix needed for latest/5-row/response features.
        zstart = start - 5
        z = np.clip((returns[zstart:stop] - mean) / scale, -8, 8)
        current = z[5:]
        earlier = z[4:-1]
        trend = pd.DataFrame(z).rolling(5, min_periods=4).mean().to_numpy()[5:]
        row_record = {'date_id': start, 'fit_last_date': start-1}
        for kind, lag in [('synchronous', 0), ('directed', 1)]:
            leader = returns[start-WINDOW-lag:start-lag]
            corr, observations = pairwise_correlation(follower, leader)
            index, weights = neighbors(corr, observations, left, right)
            impulse, fraction = aggregate_peers(current, index, weights)
            smooth, _ = aggregate_peers(trend, index, weights)
            old_impulse, _ = aggregate_peers(earlier, index, weights)
            own = np.stack([current[:, left], current[:, np.maximum(right, 0)]], axis=-1)
            gap = old_impulse - own
            for name, arr in [('available_weight', fraction), ('current_impulse', impulse), ('trend_5', smooth), ('response_gap', gap)]:
                # A genuinely absent second leg is exactly zero. Unavailable real legs stay NaN.
                a, b = arr[:, :, 0], arr[:, :, 1].copy()
                b[:, right < 0] = 0
                for suffix, v in [('difference', a-b), ('sum', a+b)]:
                    values[start:stop, :, names.index(f'close_network__{kind}_{name}_{suffix}')] = v.astype(np.float32)
            active = (np.abs(weights) > 0).sum(axis=-1)
            usable = (active[:, 0] >= MIN_PEERS) & ((right < 0) | (active[:, 1] >= MIN_PEERS))
            row_record[kind+'_target_graphs'] = int(usable.sum())
            # Training-end connectivity summary; no validation-based graph selection.
            if start <= TRAIN_STOP-1 < stop:
                unique = {}
                for i, pair in enumerate(legs):
                    for leg, anchor in enumerate(pair):
                        for j, w in zip(index[i, leg], weights[i, leg], strict=True):
                            if w:
                                unique[(anchor, assets[int(j)])] = float(w)
                by_market = {}
                for (anchor, peer), w in unique.items():
                    key = (peer.split('_')[0], anchor.split('_')[0])
                    by_market[key] = by_market.get(key, 0) + 1
                snapshots.append({'graph': kind, 'fit_last_date': start-1,
                    'edges': [{'peer': peer, 'follower': anchor, 'correlation': w} for (anchor, peer), w in sorted(unique.items())],
                    'market_edges': [{'peer_market': a, 'follower_market': b, 'edges': c} for (a,b),c in sorted(by_market.items())],
                    'note': 'Unique pair-specific incoming associations; not verified causal/economic links.'})
        refreshes.append(row_record)
        if progress and len(refreshes) % 10 == 0:
            emit('causal_network_build',last_origin=stop-1,completed_refreshes=len(refreshes))
    if np.isinf(values).any() or values.shape != (n, m, 16):
        raise Stop('Invalid network representation.')
    return values, names, {'asset_count': len(assets), 'assets': assets,
        'candidate_templates': 16, 'price_defined_targets': m,
        'refreshes': refreshes, 'training_end_networks': snapshots,
        'timing': 'All graph estimates and scales frozen before block start; all applied inputs at or before each origin.'}


def feature_audit(block, names, base_values, base_names):
    import numpy as np
    rows, known = [], {}
    def h(v):
        a = np.ascontiguousarray(v, dtype=np.float32).copy()
        a[np.isnan(a)] = np.float32(np.nan); a[a == 0] = 0
        return hashlib.sha256(a.tobytes()).hexdigest()
    for i, name in enumerate(base_names):
        known.setdefault(h(base_values[252:TRAIN_STOP, :, i]), name)
    for i, name in enumerate(names):
        a = block[252:TRAIN_STOP, :, i]; finite = np.isfinite(a); v = a[finite]
        sig = h(a)
        duplicate = known.get(sig)
        known.setdefault(sig, name)
        hist, edges = np.histogram(v, bins=24) if len(v) else ([], [])
        rows.append({'name': name, 'type': 'availability' if name in MASKS else 'numeric',
            'group': name.split('__',1)[1].split('_',1)[0],
            'training_finite_fraction': float(finite.mean()),
            'targets_with_80pct_coverage': int((finite.mean(axis=0) >= .8).sum()),
            'q05': float(np.quantile(v,.05)) if len(v) else None,
            'q95': float(np.quantile(v,.95)) if len(v) else None,
            'exact_duplicate_of': duplicate, 'training_sha256': sig,
            'histogram': {'counts': list(map(int,hist)), 'edges': list(map(float,edges))}})
    return rows

def comparison_rows(results, control, metric):
    if len(results) != 4 or {r['variant'] for r in results} != set(VARIANTS):
        raise Stop('Expected four network model results.')
    scores = {r['variant']: metric(r['metrics']['daily_rank_correlations'],180) for r in results}
    rows, eligible = [], []
    for r in results:
        name, sel = r['variant'], r['selection']
        if ((r['fold'],r['train_stop'],r['validation_start'],r['validation_stop']) != (0,1164,1169,1349)
                or r['metrics']['date_ids'] != list(range(1169,1349))
                or not math.isclose(scores[name],r['metrics']['official_metric'],rel_tol=0,abs_tol=1e-12)):
            raise Stop('Network score dates or calculation differ.')
        if (sel['candidate_templates'] != sel['retained_templates'] + sel['rejected_templates']
                or sel['retained_templates'] != len(sel['selected_names'])):
            raise Stop('Feature accounting does not balance.')
        numeric = set(sel['selected_names']).intersection(NUMERIC['synchronous']+NUMERIC['directed'])
        if name != 'network_availability' and not numeric:
            raise Stop('No numerical information was admitted; cannot claim a feature test.')
        delta = scores[name] - control['official_metric']
        extra = scores[name] - scores['network_availability']
        passes = name != 'network_availability' and delta >= GAIN and extra >= GAIN
        if passes: eligible.append(name)
        rows.append({'variant':name,'official_metric':scores[name],'delta_vs_current_market':delta,
                     'delta_vs_availability':extra,'passes_review_gate':passes,
                     'numeric_candidates':len(plan()['variants'][name])-4,'numeric_admitted':len(numeric),
                     'availability_admitted':len(set(sel['selected_names']).intersection(MASKS)),
                     'candidate_templates':sel['candidate_templates'],'retained_templates':sel['retained_templates'],
                     'rejected_templates':sel['rejected_templates']})
    return rows,eligible


def fit_one(root,u,stage,lineage,name,base,block,names,y,pairs,fold,config,report):
    import numpy as np
    import pandas as pd
    import joblib
    from commodity_prediction.domain.catalog import Panel
    from commodity_prediction.domain.model import prepare, select
    from commodity_prediction.domain.experiment import Experiment
    from commodity_prediction.domain.attribution.run import fit_stage
    chosen = plan()['variants'][name]
    added = block[:,:,[names.index(n) for n in chosen]]
    panel = Panel(np.concatenate([base.values,added],axis=2),base.dates,base.targets,base.names+chosen,dict(base.source_series))
    panel.validate()
    if stage.exists():
        raise Stop('Unexpected pre-existing model stage; recover it rather than refit automatically.')
    settings = {**config,'max_features':len(panel.names),'max_abs_correlation':1.01}
    emit('training_only_screen',variant=name,completed=len(report['results']),total=4)
    stats = prepare(panel,y,fold.train_stop,settings)
    _, audit = select(stats,Experiment(name,algorithm='histogram').candidates(panel.names),settings,False)
    if name != 'network_availability' and not set(audit['selected_names']).intersection(set(chosen)-set(MASKS)):
        raise Stop('No usable numerical network feature survives training screening; no fit for '+name)
    if set(audit['rejection_reasons']).intersection({'feature_budget','correlated','unstable_sign'}):
        raise Stop('Unreviewed screening exclusion.')
    report['fit_attempts_this_call'] += 1;u.atomic_json(report_path(root),report)
    result = fit_stage(panel,y,pairs,fold,Experiment(name,algorithm='histogram'),settings,stats,stage,lineage)
    saved = pd.read_parquet(stage/'predictions.parquet')
    replay = joblib.load(stage/'model.joblib').predict(panel,fold.validation_start,fold.validation_stop)
    error = u.exact_predictions(saved,replay)
    report['checkpoint_hashes'][name] = u.verify_stage(stage,lineage)
    report['results'].append(result);report['new_training_fits']+=1
    report['maximum_prediction_replay_error']=max(report['maximum_prediction_replay_error'],error)
    u.atomic_json(report_path(root),report)
    emit('checkpoint_verified',variant=name,completed=len(report['results']),total=4,
         official_metric=result['metrics']['official_metric'],new_fits=report['new_training_fits'])
    del panel,stats,replay;gc.collect()
    return saved


def bar_support(x,pairs):
    import numpy as np
    supported=set()
    for v in {a for p in pairs.pair for a in p.split(' - ')}:
        us=v.endswith('_adj_close');stem=v.removesuffix('close' if us else 'Close')
        if all(stem+a in x.columns for a in (['open','high','low'] if us else ['Open','High','Low'])):
            supported.add(v)
    return np.array([all(a in supported for a in v.split(' - ')) for v in pairs.pair])

def verify_complete(root,report,prior,u):
    lineage,_=identity(root,u)
    if (report.get('status') not in TERMINAL or report.get('lineage')!=lineage

            or report.get('parents_unchanged') is not True or report.get('source_unchanged') is not True
            or report.get('prefix_replay_passed') is not True or report.get('new_training_fits')!=4):
        raise Stop('Not a completed verified network study.')
    rows,eligible=comparison_rows(report['results'],report['control'],prior.metric)
    if rows!=report['comparison_rows'] or eligible!=report['candidates_for_review']:
        raise Stop('Stored network decision cannot be reproduced.')

        raise Stop('Stored decision changed.')

    with u.safe_path(out,'supervisor.lock').open('a') as lock:
        fcntl.flock(lock.fileno(),fcntl.LOCK_EX|fcntl.LOCK_NB)
        if report_path(root).exists():
            r=u.read_json(report_path(root))
            if r.get('status') not in TERMINAL:
                raise Stop('An interrupted/failed network run exists. Return the report; no automatic refitting or budget reset.')
            verify_complete(root,r,prior,u)
            emit('COMPLETE_NETWORK_STUDY_REUSED',new_fits=0)
            return {**r,'new_fits_this_notebook_call':0}
        lineage,_=identity(root,u)
        if (root/'artifacts/close_network_ablation'/lineage).exists():
            raise Stop('Existing network artifacts have no completion report; recover rather than refit.')
        if __import__('shutil').disk_usage(root).free<1024**3:raise Stop('Need 1 GiB free; do not delete old work.')
        u.atomic_json(report_path(root),{'status':'STARTING_WORKER','project':'commodity-prediction','lineage':lineage,

def private_backup(root,report,u):
    base=root/'artifacts/close_network_ablation'/report['lineage']
    paths=[base/'plan.json',base/'review.json']
    paths += [base/'features'/f for f in report['feature_checkpoint_hashes']]
    for n in VARIANTS:paths += [base/'fold_0'/n/f for f in report['checkpoint_hashes'][n]]
    pins={str(p.relative_to(root)):u.digest(p) for p in paths}
    dest=report_path(root).parent/'close_network_checkpoints.zip';temp=u.safe_path(dest.parent,dest.name+'.tmp')
    if dest.is_symlink():raise Stop('Unsafe backup destination.')
    with zipfile.ZipFile(temp,'w',zipfile.ZIP_DEFLATED,compresslevel=1) as z:
        for p in paths:z.write(p,str(p.relative_to(root)))
        z.writestr('SHA256SUMS.json',json.dumps(pins,sort_keys=True,indent=2))
    with zipfile.ZipFile(temp) as z:
        if z.testzip():raise Stop('Private archive CRC failure.')
        for name,h in pins.items():
            if hashlib.sha256(z.read(name)).hexdigest()!=h:raise Stop('Private archive content mismatch.')
    os.replace(temp,dest)
    return {'path':str(dest),'sha256':u.digest(dest),'bytes':dest.stat().st_size,'files':len(paths),
            'private':True,'raw_csvs_included':False,'contains_derived_features_models_predictions':True,'off_disk_backup_confirmed':False}

def install(root:Path=ROOT):
    root=Path(root);prior,older,u=previous(root);u.validate_source(root)
    last=read_prior(root,prior,u);validate_prior_scores(last)
    nb=notebook_document()
    files={u.safe_path(root,'scripts/commodity_close_network.py'):Path(__file__).read_bytes(),
        u.safe_path(root,'configs/manual_close_network.json'):(json.dumps(plan(),sort_keys=True,indent=2)+'\n').encode(),
        u.safe_path(root,'docs/manual_close_network.md'):protocol_markdown().encode(),
        u.safe_path(root,'notebooks/'+NOTEBOOK):(json.dumps(nb,indent=1)+'\n').encode()}
    for path,body in files.items():
        if path.exists():
            if path.suffix=='.ipynb':
                existing=u.read_json(path)
                if [(c['cell_type'],c.get('source')) for c in existing['cells']] != [(c['cell_type'],c['source']) for c in nb['cells']]:
                    raise Stop('Existing notebook edits preserved: '+str(path))
            elif path.read_bytes()!=body:raise Stop('Existing different file preserved: '+str(path))
    for path,body in files.items():
        if not path.exists():u.write_new(path,body)
    print('RESULT: CLOSE_NETWORK_NOTEBOOK_READY\nNOTEBOOK: '+str(root/'notebooks'/NOTEBOOK)+'\nKERNEL: Commodity - manual (verified)',flush=True)
    return {'new_fits':0,'files':[str(p) for p in files]}


def main():
    p=argparse.ArgumentParser(description=__doc__);p.add_argument('--root',type=Path,default=ROOT)
    p.add_argument('--worker',action='store_true',help=argparse.SUPPRESS);a=p.parse_args()
    try:
        if a.worker:worker(a.root)
        else:install(a.root)
    except BaseException as exc:
        print('RESULT: STOPPED\nERROR: '+type(exc).__name__+': '+str(exc),flush=True)
        print('Preserve error/report and stop space. Do not repeat unchanged. No automatic shutdown.',flush=True)
        return 2
    return 0


if __name__=='__main__':raise SystemExit(main())
