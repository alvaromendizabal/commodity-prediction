def fixture(T=160,J=16):
    rng=np.random.default_rng(20260912)
    names=['released_priors__'+s for s in m.PRIOR_NAMES]
    values=rng.normal(0,.03,(T,J,len(names))).astype(np.float32)
    for k,s in enumerate(m.PRIOR_NAMES):
        if s.startswith('risk_'):values[:,:,k]=np.abs(values[:,:,k])+.02
    pairs=pd.DataFrame({'target':[f'target_{j}' for j in range(J)],'lag':np.tile([1,2,3,4],J//4)})
    y=rng.normal(0,.03,(T,J));risk=np.full((T,J),.05)
    return values,names,pairs,y,risk

class EventHistoryTests(unittest.TestCase):
    def setUp(self):self.a,self.names,self.pairs,self.y,self.risk=fixture()
    def test_24_event_features(self):
        c,n,meta=m.event_block(self.y,self.risk,self.pairs)
        self.assertEqual(c.shape,(160,16,24));self.assertEqual(len(set(n)),24)
        self.assertEqual([len(x) for x in m.groups(17,meta).values()],[6,15,15,24])
    def test_event_group_union(self):
        c,n,meta=m.event_block(self.y,self.risk,self.pairs);g=m.groups(17,meta)
        self.assertEqual(set(g['event_joint']),set(g['event_location'])|set(g['event_shape']))
        self.assertEqual(set(g['event_location'])&set(g['event_shape']),set(g['event_clock_only']))
    def test_release_boundary_each_horizon(self):
        for j,h in enumerate([1,2,3,4]):
            changed=self.y.copy();changed[90,j]=100
            a,_,_=m.event_block(self.y,self.risk,self.pairs);b,_,_=m.event_block(changed,self.risk,self.pairs)
            with self.subTest(horizon=h):
                np.testing.assert_array_equal(a[:90+h+1,j],b[:90+h+1,j])
                self.assertFalse(np.array_equal(a[90+h+1,j],b[90+h+1,j],equal_nan=True))
    def test_observed_event_not_calendar_window(self):
        self.y[:,0]=np.nan;origin=np.arange(0,150,3);self.y[origin,0]=np.arange(len(origin))/100
        c,n,_=m.event_block(self.y,self.risk,self.pairs);t=100;valid=origin[origin+2<=t][-8:]
        expected=self.y[valid,0].mean()/.05
        self.assertAlmostEqual(c[t,0,n.index('event_history17__k8__mean_over_risk')],expected,places=5)
    def test_event_mean_independent_reference(self):
        c,n,_=m.event_block(self.y,self.risk,self.pairs)
        for j in [0,3,9]:
            t=120;cut=t-int(self.pairs.iloc[j]['lag'])-1
            for k in [8,21,63]:
                z=self.y[:cut+1,j];z=z[np.isfinite(z)][-k:]
                with self.subTest(target=j,k=k):self.assertAlmostEqual(c[t,j,n.index(f'event_history17__k{k}__mean_over_risk')],z.mean()/.05,places=5)
    def test_event_median_independent_reference(self):
        c,n,_=m.event_block(self.y,self.risk,self.pairs);t=120;j=2;k=21;cut=t-4
        self.assertAlmostEqual(c[t,j,n.index(f'event_history17__k{k}__median_over_risk')],np.median(self.y[cut-k+1:cut+1,j])/.05,places=5)
    def test_event_shape_is_scale_invariant(self):
        a,n,meta=m.event_block(self.y,self.risk,self.pairs);b,_,_=m.event_block(self.y*7,self.risk*7,self.pairs)
        np.testing.assert_allclose(a,b,atol=2e-6,rtol=2e-6,equal_nan=True)
    def test_event_future_perturbation(self):

class RealRepositorySmoke(unittest.TestCase):
    def test_tiny_repository_model_and_exact_replay(self):
        """One deliberately tiny synthetic fit, separate from private experiment fits."""
        import joblib
        from threadpoolctl import threadpool_limits
        from commodity_prediction.domain.catalog import Panel
        from commodity_prediction.domain.model import prepare,select,fit
        from commodity_prediction.domain.experiment import Experiment
        rng=np.random.default_rng(42);T,J,P=100,4,3
        a=rng.normal(size=(T,J,P)).astype(np.float32);names=['synthetic__a','synthetic__b','synthetic__c']
        panel=Panel(a,list(range(T)),[f'target_{j}' for j in range(J)],names,{})
        y=pd.DataFrame(.01*a[:,:,0]+rng.normal(0,.01,(T,J)),index=panel.dates,columns=panel.targets)
        config=json.loads((ROOT/'configs/domain_study.json').read_text())
        config.update(warmup_dates=5,max_features=P,max_abs_correlation=1.01,histogram_iterations=2,histogram_min_samples_leaf=8)
        with threadpool_limits(limits=1):
            stats=prepare(panel,y,70,config);sel,_=select(stats,Experiment('tiny',algorithm='histogram').candidates(names),config,False)
            self.assertEqual(len(sel),3)
            model=fit(stats,sel,panel,y,'histogram',config)
            self.assertIsNotNone(model.estimator)
            p=model.predict(panel,75,90)
            self.assertTrue(np.isfinite(p.to_numpy()).all())
            with tempfile.TemporaryDirectory() as d:
                path=Path(d)/'model.joblib';joblib.dump(model,path)
                pd.testing.assert_frame_equal(p,joblib.load(path).predict(panel,75,90),check_exact=True)
    def test_official_metric_independent_known_example(self):
        from commodity_prediction.metrics import daily_rank_correlations, correlation_sharpe
        truth=pd.DataFrame([[1.,2.,3.],[1.,2.,3.],[1.,2.,3.],[1.,3.,2.]],columns=['a','b','c'])
        prediction=pd.DataFrame([[1.,2.,3.],[3.,2.,1.],[1.,3.,2.],[1.,2.,3.]],columns=['a','b','c'])
        daily=daily_rank_correlations(truth,prediction)
        np.testing.assert_allclose(np.asarray(daily),[1.,-1.,.5,.5],atol=1e-12)
        self.assertAlmostEqual(correlation_sharpe(daily),1/3,places=12)
    def test_real_parquet_write_and_read(self):
        df=pd.DataFrame({'target_0':[.2,.3],'target_1':[.4,.5]},index=pd.Index([1,2],name='date_id'))
        with tempfile.TemporaryDirectory() as d:
            p=Path(d)/'p.parquet';df.to_parquet(p);pd.testing.assert_frame_equal(df,pd.read_parquet(p),check_exact=True)
